\begin{frame}{MED = Minimal Dimension to achieve (robust) $k$-shattering}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.55\textwidth}
      \begin{block}{$k$-shattering}
        Store $m$ objects as vectors.  For every answer set
        $S\subseteq X$, $1\le |S|\le k$, choose a query and threshold:
        \[
          \min_{i\in S}s(v_i,u_S)
          >
          b_S
          \ge
          \max_{j\notin S}s(v_j,u_S).
        \]
        If $|S|$ is known, this is top-$|S|$ retrieval.
      \end{block}

      \begin{block}{Robust $k$-shattering}
        Unit-normalized objects and queries must also have margin:
        \[
          \min_{i\in S}s(v_i,u_S)
          \ge
          \max_{j\notin S}s(v_j,u_S)+\epsilon.
        \]
      \end{block}
    \end{column}
    \begin{column}{0.40\textwidth}
      \centering
      \begin{tikzpicture}[x=1cm,y=1cm]
        \draw[->,line width=0.4pt] (0,0.2) -- (0,4.0);
        \node[anchor=east] at (-0.05,3.85) {\small score};
        \draw[dashed,blue!55!black] (-0.45,2.1) -- (2.9,2.1)
        node[right] {\small $b_S$};
        \draw[densely dotted,red!70!black] (-0.45,2.75) -- (2.9,2.75)
        node[right] {\small $b_S+\epsilon$};
        \draw[<->,red!70!black] (2.55,2.1) -- (2.55,2.75)
        node[midway,right] {\scriptsize gap};

        \foreach \x/\y in {0.55/3.45,1.35/3.15,2.05/3.35}
        \fill[blue!70!black] (\x,\y) circle (3pt);
        \foreach \x/\y in {0.45/1.1,1.15/1.55,1.9/0.8,2.35/1.4}
        \fill[gray!60] (\x,\y) circle (3pt);
        \node[blue!70!black] at (1.35,3.78) {\small selected $S$};
        \node[gray!60!black] at (1.55,0.35) {\small unselected};

        \node[draw,rounded corners,inner sep=4pt,align=center]
        at (1.25,4.45) {\small one query\\[-0.2em]\small ranks all objects};
      \end{tikzpicture}

      \vspace{0.7em}
      \small
      Exact $k$-shattering asks for strict separation.
      Robust $k$-shattering asks for a normalized score gap that cannot be removed by rescaling.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{MED bounds}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Upper bound construction: \\cyclic polytope and square polynomials}
        Pick $t_1<t_2 <\cdots < t_m$. Put objects in $\R^{2k}$:
        \[
          v_i=(t_i,t_i^2,\ldots,t_i^{2k}).
        \]
        For target $S$, define
        \[
          P_S(t)=\prod_{i\in S}(t-t_i).
        \]
        Use coefficients of $-P_S(t)^2$ as $u_S$.

        \vspace{0.2em}
        Then selected objects score $0$, and every unselected object
        scores $<0$.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Lower bound: VC dimension}
        MED realizes every subset of any $k$ chosen objects, so the score
        threshold class shatters $k$ points.

        \vspace{0.3em}
        Linear thresholds in $\R^d$ have VC dimension $d+1$, hence
        \[
          d+1\ge k\quad\Rightarrow\quad d\ge k-1.
        \]
      \end{block}

      \vspace{0.2em}
      \centering
      \small
      \begin{tabular}{lcc}
        \toprule
        Scoring & Lower & Upper \\
        \midrule
        Inner product & $k-1$ & $2k$ \\
        Euclidean & $k-1$ & $2k$ \\
        Cosine & $k-1$ & $2k+1$ \\
        \bottomrule
      \end{tabular}
    \end{column}
  \end{columns}

  \vspace{0.5em}
  \centering
  \large Exact MED is $\Theta(k)$ and does not grow with $m$.
\end{frame}

\begin{frame}{RMED: feasible margins and Gaussian upper bound}
  \small
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Feasibility of $\epsilon$ (informal).}

      Every robust witness must satisfy
      \[
        \epsilon\le
        \epsilon_\star(m,k)
        =
        \frac{m}{\sqrt{k(m-1)(m-k)}}.
      \]

      \begin{itemize}
        \item $\epsilon>\epsilon_\star(m,k)$: infeasible.
        \item $\epsilon\le\epsilon_\star(m,k)$: feasible in $\R^{m-1}$
          by a regular simplex.
        \item If $m/k\to\infty$, then
          $\epsilon_\star(m,k)\sim1/\sqrt{k}$.
      \end{itemize}
    \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Gaussian centroid witness.}

      We sample random unit object vectors $v_i$ and construct query vectors
      \[
        u_S=
        \frac{\sum_{i\in S}v_i}
        {\left\|\sum_{i\in S}v_i\right\|_2}.
      \]
    \end{block}
      At the feasible margin scale $\epsilon_k=c/\sqrt{k}$, when $d=Ck^2\log m$,
      concentration and union bounds implies that robust MED exists.
      \[
        \RMEDC(m,k,\epsilon_k)\le O(k^2\log m),
      \]
      and $\RMED$ is no larger than this restricted centroid-query quantity.

    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Conclusion}
  \begin{block}{Exact MED}
    Cyclic-polytopal neighborliness gives explicit $\Theta(k)$-dimensional
    witnesses for arbitrary top-$k$ answer sets.
  \end{block}

  \begin{block}{Robust MED}
    Normalized margins change the regime: feasible gaps are capped by
    $\epsilon_\star(m,k)$, and Gaussian centroids give
    $O(k^2\log m)$ dimensions at the $c/\sqrt{k}$ scale.
  \end{block}

  \begin{block}{Empirical interpretation}
    The observed failures are not exact-capacity failures.  The remaining
    difficulties lie in learning,
    tokenization, objectives, conditioning, finite precision, and optimization.
  \end{block}
\end{frame}
