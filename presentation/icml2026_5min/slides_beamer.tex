\begin{frame}{The approximability question}
  Embedding retrieval stores objects as vectors, embeds a query, and
  retrieves by score comparison.

  \vspace{0.5em}
  \begin{block}{Minimal Embeddable Dimension}
    Given $m$ objects, $\MED$ is the least $d$ such that some vectors in
    $\R^d$ can exactly retrieve every answer set
    \[
      S \subseteq X,\qquad 1 \leq |S| \leq k
    \]
    by score comparison and a threshold.
  \end{block}

  This isolates exact geometric capacity from positive-margin or learned
  embedding protocols.
\end{frame}

\begin{frame}{Exact capacity is $\Theta(k)$, not a function of $m$}
  \begin{center}
  \begin{tabular}{lcc}
    \toprule
    Scoring rule & Lower bound & Upper bound \\
    \midrule
    Inner product & $k-1$ & $2k$ \\
    Euclidean distance & $k-1$ & $2k$ \\
    Cosine similarity & $k-1$ & $2k+1$ \\
    \bottomrule
  \end{tabular}
  \end{center}

  \vspace{0.8em}
  \begin{itemize}
    \item Lower bounds come from VC dimension.
    \item Upper bounds are explicit constructions.
    \item Exact geometric capacity does not grow with the universe size $m$.
  \end{itemize}
\end{frame}

\begin{frame}{The construction in one line}
  Place objects on the moment curve:
  \[
    v_i = (t_i,t_i^2,\ldots,t_i^{2k}).
  \]

  For a target subset $S$, form
  \[
    P_S(t)=\prod_{i\in S}(t-t_i).
  \]

  Use the coefficients of $-P_S(t)^2$ as the query vector.

  \vspace{0.6em}
  \begin{block}{Closed-form witness}
    Selected objects score at the shared maximum; every other object scores lower.
  \end{block}
\end{frame}

\begin{frame}{Robust MED: normalized gaps change the regime}
  Exact MED is the no-margin problem.

  \vspace{0.4em}
  RMED adds unit-norm vectors and a fixed selected-unselected score gap
  $\epsilon$.

  \vspace{0.5em}
  Positive gap makes $m$ reappear through packing:
  \[
    \RMED(m,k,\epsilon)
    \geq
    \frac{\log \binom{m}{k}}{\log(1+2/\epsilon)} .
  \]

  \vspace{0.2em}
  Finite-$m$ gaps are capped by
  \[
    \epsilon_\star(m,k)=\frac{m}{\sqrt{k(m-1)(m-k)}}.
  \]

  \vspace{0.1em}
  The paper's RMED shorthand is the large-universe regime
  $m/k\to\infty$, where $\epsilon_\star(m,k)\sim1/\sqrt{k}$.

  \vspace{0.2em}
  At the feasible $c/\sqrt{k}$ scale, a Gaussian centroid witness gives
  \[
    \RMEDC(m,k,c/\sqrt{k}) \leq O(k^2\log m).
  \]
\end{frame}

\begin{frame}{What this means for retrieval}
  \begin{block}{Exact MED}
    Cyclic-polytopal neighborliness gives $\Theta(k)$ dimensions for inner
    product, cosine, and Euclidean scoring, independent of $m$ up to constants.
  \end{block}

  \begin{block}{Robust MED}
    The normalized-gap problem is different: finite-$m$ gaps are capped by
    $\epsilon_\star(m,k)$, and Gaussian centroid witnesses give
    $O(k^2\log m)$ dimensions at the feasible large-universe scale.
  \end{block}

  \begin{block}{Empirical interpretation}
    The synthetic and LIMIT failures are not failures of geometric
    approximability; the remaining difficulties lie in learning,
    tokenization, objectives, conditioning, finite precision, and optimization.
  \end{block}
\end{frame}
