    \sectionhead{teal}{1. Definitions}
    {\MiniHeadFont\color{teal} MED is a dimension of a score family, not a
    training protocol.}\par\vspace{0.12in}
    Store $m$ objects as $X=\{v_1,\ldots,v_m\}\subset\mathbb{R}^d$.
    For every answer set $S\subseteq X$ with $1\le |S|\le k$, a query
    vector $u_S$ and threshold $b_S$ must retrieve exactly $S$.

    \vspace{0.18in}
    \begin{posterbox}[colback=posterbg, height=8.2in]
      {\SubheadFont What must be realized?}\par\vspace{0.08in}
      \begin{minipage}[t]{0.47\linewidth}

        \begin{center}

    \vspace{0.25in}
    \begin{posterbox}[colback=teallight, colframe=teal, height=4.15in]
      {\SubheadFont Exact MED}\par\vspace{0.06in}
      \[
        \min_{i\in S}s(v_i,u_S)>b_S>
        \max_{j\notin S}s(v_j,u_S).
      \]
      {\SmallFont Strict ordering is enough. No prescribed margin or
      normalization is required.}
    \end{posterbox}

    \vspace{0.25in}
    \begin{posterbox}[colback=amberlight, colframe=amber, height=4.25in]
      {\SubheadFont Robust MED}\par\vspace{0.06in}
      \[
        \min_{i\in S}s(v_i,u_S)\ge
        \max_{j\notin S}s(v_j,u_S)+\epsilon .
      \]
      {\SmallFont Objects and queries are normalized. A uniform positive
      score gap must work for every target set.}
    \end{posterbox}

    \vspace{0.25in}
    \begin{posterbox}[colback=bluelight, colframe=blue, height=5.35in]
      {\SubheadFont VC lower bound}\par
      If all subsets of size at most $k$ can be realized, then the
      thresholded score class shatters any chosen $k$ objects. Thus
      \[
        k\le \mathrm{VCD}(\mathcal{C}_{s,d}).
      \]
      For inner-product and Euclidean threshold families,
      $\mathrm{VCD}=d+1$, so exact MED and every robust MED satisfy
      $d\ge k-1$.
    \end{posterbox}

    \vfill
    \begin{posterbox}[colback=teallight, colframe=teallight, height=1.45in, valign=center]
      \centering{\SubheadFont Lower bounds are $\Omega(k)$; the question is whether $m$ must enter.}
    \end{posterbox}
  \end{posterbox}

\begin{minipage}[t]{0.315\linewidth}
  \begin{posterbox}[height=29.35in]
    \sectionhead{amber}{2. Theory}
    {\MiniHeadFont\color{amber} Exact capacity is $m$-independent; robust
    capacity pays for margin.}\par\vspace{0.12in}

    \begin{posterbox}[colback=teallight, colframe=teal, height=3.35in]
      {\SubheadFont Exact MED theorem}\par\vspace{0.08in}
      {\TableFont
      \renewcommand{\arraystretch}{1.2}
      \begin{tabular}{>{\bfseries}lcc}
        \toprule
        Score & Lower & Upper \\
        \midrule
        Inner product & $k-1$ & $2k$ \\
        Euclidean & $k-1$ & $2k$ \\
        Cosine & $k-1$ & $2k+1$ \\
        \bottomrule
      \end{tabular}}\par\vspace{0.08in}
      {\SmallFont Main message: $\mathrm{MED}(m,k)=\Theta(k)$, independent
      of $m$ up to constants.}
    \end{posterbox}

    \vspace{0.22in}
    \begin{posterbox}[colback=posterbg, height=6.85in]
      {\SubheadFont Cyclic-polytopal witness}\par
      Put all $m$ objects on the moment curve in $\mathbb{R}^{2k}$:
      \[
        v_i=(t_i,t_i^2,\ldots,t_i^{2k}).
      \]
      For target $S$, use the square polynomial
      \[
        P_S(t)=\prod_{i\in S}(t-t_i),\qquad
        \langle v(t),q_S\rangle=c_0-P_S(t)^2 .
      \]
      Selected points are exactly the roots of $P_S$ and score $c_0$;
      every outsider scores below $c_0$.
      \begin{center}
      \begin{tikzpicture}[x=1in,y=1in,scale=0.92]
        \draw[gray!55,line width=1.5pt] (0,0) -- (7.2,0);
        \draw[blue,line width=4pt,line cap=round]
          (0.2,0.35) .. controls (1.25,1.55) and (2.4,1.35) .. (3.2,0.7)
          .. controls (4.2,-0.15) and (5.8,-0.05) .. (6.95,0.95);
        \foreach \x/\y/\c in {0.55/0.55/blue,1.55/1.42/amber,2.9/0.84/blue,4.25/-0.08/amber,6.2/0.78/blue}
          \fill[\c] (\x,\y) circle (0.12);
        \draw[amber,line width=3pt] (1.32,1.85) -- (1.55,2.16) -- (1.78,1.85);
        \draw[amber,line width=3pt] (4.02,-0.42) -- (4.25,-0.73) -- (4.48,-0.42);
      \end{tikzpicture}
      \end{center}
    \end{posterbox}

    \vspace{0.22in}
    \begin{posterbox}[colback=amberlight, colframe=amber, height=3.75in]
      {\SubheadFont Feasible robust margin}\par\vspace{0.04in}
      \[
        \epsilon_\star(m,k)=
        \frac{m}{\sqrt{k(m-1)(m-k)}}
        \sim \frac{1}{\sqrt{k}} .
      \]
      If $\epsilon>\epsilon_\star(m,k)$, robust retrieval is impossible
      in any finite dimension. The ceiling is tight in $\mathbb{R}^{m-1}$
      by a regular simplex.
    \end{posterbox}

    \vspace{0.22in}
    \begin{posterbox}[colback=amberlight, colframe=amberlight, height=5.45in]
      {\SubheadFont Gaussian centroid upper bound}\par
      Sample random unit object vectors and query by normalized selected
      centroid:
      \[
        u_S=\frac{\sum_{i\in S}v_i}
        {\left\|\sum_{i\in S}v_i\right\|_2}.
      \]
      Selected objects get a self-correlation term; outsiders contribute
      coherence noise. With positive probability,
      \[
        d=Ck^2\log m
      \]
      robustly retrieves every at-most-$k$ subset with margin
      $\Omega(1/\sqrt{k})$.
    \end{posterbox}

    \vspace{0.22in}
    \begin{posterbox}[colback=bluelight, colframe=blue, height=2.3in]
      {\SubheadFont Score-rule transfer}\par
      Cosine agrees with inner product after normalization; Euclidean
      squared-distance comparisons differ by
      $2\langle u_S,v_i-v_j\rangle$.
    \end{posterbox}

    \vfill
    \begin{posterbox}[colback=amberlight, colframe=amberlight, height=1.25in, valign=center]
      \centering{\SubheadFont Robustness reintroduces $m$, but the Gaussian witness pays only $\log m$.}
    \end{posterbox}
  \end{posterbox}
